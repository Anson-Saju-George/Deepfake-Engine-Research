# requirements.txt — deepfake-detection pipeline (data -> train -> evaluate)
#
# Human-readable dependency list for the ACTIVE code surface (data/, proc/, train/,
# proc/code_snippets/). Unpinned = installs latest. For the exact versions the
# performance benchmarks were certified on, use requirements.lock.txt (a curated,
# project-scoped lock matching the /opt/ml env — torch 2.11.0+cu129, cuDNN 9.17.1.4).
#
# GPU note: torch/torchvision are CUDA builds. Install them from the PyTorch index:
#   pip install torch torchvision --index-url https://download.pytorch.org/whl/cu129
# Graphing deps (matplotlib, pandas) are intentionally EXCLUDED — graphs/ runs in its
# own separate .graph-venv, kept out of the training environment on purpose.

# ---- core numerical / imaging ----
numpy
Pillow
opencv-python                 # cv2 — video decode + frame color conversion (data/dataloader.py, proc/*)
                              # headless WSL: swap for opencv-python-headless (same import name)

# ---- deep learning ----
torch                         # install from the cu129 index (see GPU note above)
torchvision
timm                          # backbone zoo (ConvNeXt / Swin / ViT / etc.)

# ---- metrics / evaluation ----
scikit-learn                  # sklearn.metrics — train/eval_predictions_common.py, proc/code_snippets/metrics.py
scipy                         # scipy.stats.wilcoxon — proc/code_snippets/metrics.py

# ---- utilities ----
tqdm
psutil                        # worker calibration — proc/pre/calibrate_workers.py

# decord
# av

# ---- optional: model profiling (proc/code_snippets/profile_model_complexity.py) ----

# fvcore
